import React, { useState } from 'react';
import { View, Text, TextInput, Button, FlatList, StyleSheet, ScrollView } from 'react-native';

export default function App() {
  const [product, setProduct] = useState('');
  const [qty, setQty] = useState('');
  const [price, setPrice] = useState('');

  const [stockName, setStockName] = useState('');
  const [stockQty, setStockQty] = useState('');

  const [sales, setSales] = useState([]);
  const [stock, setStock] = useState([]);

  const addSale = () => {
    if (!product || !qty || !price) return;
    const total = parseFloat(qty) * parseFloat(price);
    setSales([...sales, { product, qty, price, total }]);
    setProduct('');
    setQty('');
    setPrice('');
  };

  const addStock = () => {
    if (!stockName || !stockQty) return;
    setStock([...stock, { name: stockName, qty: stockQty }]);
    setStockName('');
    setStockQty('');
  };

  const totalSales = sales.reduce((sum, item) => sum + item.total, 0);

  return (
    <ScrollView style={styles.container}>

      <Text style={styles.title}>SIMAMIA BIASHARA</Text>

      <View style={styles.card}>
        <Text>Mauzo ya Leo: {totalSales} TZS</Text>
        <Text>Stoo: {stock.length} bidhaa</Text>
      </View>

      <View style={styles.card}>
        <Text style={styles.header}>Mauzo</Text>
        <TextInput placeholder="Jina la bidhaa" value={product} onChangeText={setProduct} style={styles.input} />
        <TextInput placeholder="Idadi" value={qty} onChangeText={setQty} keyboardType="numeric" style={styles.input} />
        <TextInput placeholder="Bei" value={price} onChangeText={setPrice} keyboardType="numeric" style={styles.input} />
        <Button title="Ongeza Mauzo" onPress={addSale} />
      </View>

      <View style={styles.card}>
        <Text style={styles.header}>Stoo</Text>
        <TextInput placeholder="Jina la bidhaa" value={stockName} onChangeText={setStockName} style={styles.input} />
        <TextInput placeholder="Idadi" value={stockQty} onChangeText={setStockQty} keyboardType="numeric" style={styles.input} />
        <Button title="Ongeza Stoo" onPress={addStock} />
      </View>

      <View style={styles.card}>
        <Text style={styles.header}>Orodha ya Mauzo</Text>
        {sales.map((item, index) => (
          <Text key={index}>
            {item.product} - {item.qty} x {item.price} = {item.total}
          </Text>
        ))}
      </View>

    </ScrollView>
  );
}

const styles = StyleSheet.create({
  container: { padding: 15, marginTop: 30 },
  title: { fontSize: 22, fontWeight: 'bold', textAlign: 'center', marginBottom: 10 },
  card: { padding: 10, borderWidth: 1, marginBottom: 10, borderRadius: 8 },
  header: { fontWeight: 'bold', marginBottom: 5 },
  input: { borderWidth: 1, marginBottom: 5, padding: 8, borderRadius: 5 }
});
